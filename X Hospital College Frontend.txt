  <script>
    // Data structure for states -> cities -> areas as requested
    const data = {
      maharashtra: {
        mumbai: ['Bandra','Andheri','Dadar'],
        pune: ['Kothrud','Sivajinagar','Hadaspar'],
        nagpur: ['Civil Line','Sitabuldi','Dharampeth']
      },
      gujarat: {
        vadodara: ['Alkapuri','Gotri','Manjalpur'],
        ahmedabad: ['Navrangpura','Paldi','Vastrapur'],
        surat: ['Adajan','Udhna','Varchana']
      },
      telangana: {},
      tamilnadu: {},
      madhya_pradesh: {},
      jammu_and_kashmir: {},
      gujarat_full: {},
      others: {}
    };

    // Utility to populate select
    function fillSelect(selectEl, items, placeholder) {
      selectEl.innerHTML = '';
      const opt0 = document.createElement('option');
      opt0.value = '';
      opt0.textContent = placeholder || '-- Select --';
      selectEl.appendChild(opt0);
      for (const key of Object.keys(items)) {
        const opt = document.createElement('option');
        opt.value = key;
        opt.textContent = capitalize(key.replace(/_/g,' '));
        selectEl.appendChild(opt);
      }
    }
    function fillArraySelect(selectEl, arr, placeholder) {
      selectEl.innerHTML = '';
      const opt0 = document.createElement('option'); opt0.value=''; opt0.textContent = placeholder||'-- Select --'; selectEl.appendChild(opt0);
      arr.forEach(a=>{const o=document.createElement('option');o.value=a;o.textContent=a;selectEl.appendChild(o);});
    }
    function capitalize(s){return s.charAt(0).toUpperCase()+s.slice(1)}

    // Elements
    const stateHosp = document.getElementById('state-hosp');
    const cityHosp = document.getElementById('city-hosp');
    const areaHosp = document.getElementById('area-hosp');
    const searchHosp = document.getElementById('search-hosp');
    const resultHosp = document.getElementById('result-hosp');
    const clearHosp = document.getElementById('clear-hosp');

    const stateClinic = document.getElementById('state-clinic');
    const cityClinic = document.getElementById('city-clinic');
    const areaClinic = document.getElementById('area-clinic');
    const searchClinic = document.getElementById('search-clinic');
    const resultClinic = document.getElementById('result-clinic');
    const clearClinic = document.getElementById('clear-clinic');

    // Populate state selects for hospital (7 states) - already in HTML but we want consistent keys
    // For clinic (two states) already in HTML.

    // When hospital state changes
    stateHosp.addEventListener('change',()=>{
      const s = stateHosp.value;
      if(!s){ cityHosp.innerHTML='<option value="">-- Select City --</option>'; cityHosp.disabled=true; areaHosp.disabled=true; areaHosp.innerHTML='<option value="">-- Select Area --</option>'; return; }
      if(s==='maharashtra' || s==='gujarat'){
        const cities = data[s];
        fillSelect(cityHosp, cities, '-- Select City --');
        cityHosp.disabled=false; areaHosp.disabled=true; areaHosp.innerHTML='<option value="">-- Select Area --</option>';
      } else {
        // no city data for other states in demo
        cityHosp.innerHTML='<option value="">Not available in demo</option>';
        cityHosp.disabled=true; areaHosp.disabled=true; areaHosp.innerHTML='<option value="">--</option>';
      }
    });

    cityHosp.addEventListener('change',()=>{
      const s = stateHosp.value; const c = cityHosp.value;
      if(!c){ areaHosp.disabled=true; areaHosp.innerHTML='<option value="">-- Select Area --</option>'; return; }
      const areas = data[s][c] || [];
      fillArraySelect(areaHosp, areas, '-- Select Area --'); areaHosp.disabled=false;
    });

    searchHosp.addEventListener('click',()=>{
      const s = stateHosp.value; const c = cityHosp.value; const a = areaHosp.value;
      if(!s){ alert('Please select a state first.'); return; }
      let txt = `Results (demo): State: ${capitalize(s.replace(/_/g,' '))}`;
      if(c) txt += ` • City: ${capitalize(c)}`;
      if(a) txt += ` • Area: ${a}`;
      resultHosp.textContent = txt;
      // For presentation, also show a modal-like highlight (simple)
      resultHosp.style.border='1px solid rgba(14,165,164,0.18)';
    });

    clearHosp.addEventListener('click',()=>{
      stateHosp.value=''; cityHosp.innerHTML='<option value="">-- Select City --</option>'; cityHosp.disabled=true; areaHosp.innerHTML='<option value="">-- Select Area --</option>'; areaHosp.disabled=true; resultHosp.textContent='Choose state → city → area and click Search Hospitals.'; resultHosp.style.border='1px dashed rgba(3,7,18,0.04)';
    });

    // Clinic flows (only Maharashtra & Gujarat)
    stateClinic.addEventListener('change',()=>{
      const s = stateClinic.value;
      if(!s){ cityClinic.innerHTML='<option value="">-- Select City --</option>'; cityClinic.disabled=true; areaClinic.disabled=true; areaClinic.innerHTML='<option value="">-- Select Area --</option>'; return; }
      const cities = data[s];
      fillSelect(cityClinic, cities, '-- Select City --'); cityClinic.disabled=false; areaClinic.disabled=true; areaClinic.innerHTML='<option value="">-- Select Area --</option>';
    });
    cityClinic.addEventListener('change',()=>{
      const s = stateClinic.value; const c = cityClinic.value;
      if(!c){ areaClinic.disabled=true; areaClinic.innerHTML='<option value="">-- Select Area --</option>'; return; }
      const areas = data[s][c] || [];
      fillArraySelect(areaClinic, areas, '-- Select Area --'); areaClinic.disabled=false;
    });
    searchClinic.addEventListener('click',()=>{
      const s = stateClinic.value; const c = cityClinic.value; const a = areaClinic.value;
      if(!s){ alert('Please choose a state first.'); return; }
      let txt = `Clinic results (demo): State: ${capitalize(s)}`;
      if(c) txt += ` • City: ${capitalize(c)}`;
      if(a) txt += ` • Area: ${a}`;
      resultClinic.textContent = txt; resultClinic.style.border='1px solid rgba(14,165,164,0.18)';
    });
    clearClinic.addEventListener('click',()=>{
      stateClinic.value=''; cityClinic.innerHTML='<option value="">-- Select City --</option>'; cityClinic.disabled=true; areaClinic.innerHTML='<option value="">-- Select Area --</option>'; areaClinic.disabled=true; resultClinic.textContent='Pick state and city — then search.'; resultClinic.style.border='1px dashed rgba(3,7,18,0.04)';
    });

    // Initialize city selects disabled
    cityHosp.disabled=true; areaHosp.disabled=true; cityClinic.disabled=true; areaClinic.disabled=true;

    // Pre-fill hospital states' city lists for the two known states so HTML select keys match data keys
    // For demo we want the option values to match data keys (mumbai,pune,nagpur etc.). We'll patch options when state selected.

    // Small helper to set city select options from object keys
    function setCityOptions(selectEl, obj){
      selectEl.innerHTML='';
      const defaultOpt = document.createElement('option'); defaultOpt.value=''; defaultOpt.textContent='-- Select City --'; selectEl.appendChild(defaultOpt);
      Object.keys(obj).forEach(k=>{const o=document.createElement('option');o.value=k;o.textContent=capitalize(k);selectEl.appendChild(o)});
    }

    // Make sure the state-hosp select option values map to our data keys
    // (HTML already uses 'maharashtra' and 'gujarat') — nothing to do.

    // For presentation: highlight the two clinic states visually when page loads
    window.addEventListener('load',()=>{
      // animate little badge
      const badge = document.querySelector('.badge');
      badge.style.transform='translateY(-2px)';
    });

  </script>
